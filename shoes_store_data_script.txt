create database shoes_store;
use shoes_store;

CREATE TABLE shoes (
    shoes_id INT AUTO_INCREMENT PRIMARY KEY,
    shoes_name VARCHAR(100) NOT NULL,
    shoes_price DECIMAL(10, 2) NOT NULL,
    shoes_brand VARCHAR(50) NOT NULL,
    shoes_description VARCHAR(255),
    shoes_img  VARCHAR(255)
);
